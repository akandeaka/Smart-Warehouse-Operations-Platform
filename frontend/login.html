/* ========================================== AKINEX SWOP Enterprise Login login.js - Complete Authentication ========================================== */ document.addEventListener("DOMContentLoaded", function() { // ============================================================ // API Configuration // ============================================================ const API_URL = 'https://warehouse-backend-new-u9rq.onrender.com/api'; // ============================================================ // DOM Elements // ============================================================ const loginForm = document.getElementById('loginForm'); const loginBtn = document.getElementById('loginBtn'); const username = document.getElementById('username'); const password = document.getElementById('password'); const twofaCode = document.getElementById('twofaCode'); const togglePassword = document.getElementById('togglePassword'); const themeToggle = document.getElementById('themeToggle'); const portalBadge = document.getElementById('portalBadge'); const errorMsg = document.getElementById('errorMsg'); const infoMsg = document.getElementById('infoMsg'); const rememberMe = document.getElementById('rememberMe'); const twofaGroup = document.getElementById('twofaGroup'); // ============================================================ // State Variables // ============================================================ let redirectPortal = 'sales'; let twofaSessionId = null; let isTwoFAMode = false; // ============================================================ // 1. PORTAL DETECTION // ============================================================ const urlParams = new URLSearchParams(window.location.search); const redirectParam = urlParams.get('redirect'); if (redirectParam === 'super-admin') { redirectPortal = 'super-admin'; } else if (redirectParam === 'manager-dashboard') { redirectPortal = 'manager'; } else { const referrer = document.referrer; if (referrer.includes('super-admin')) { redirectPortal = 'super-admin'; } else if (referrer.includes('manager-dashboard')) { redirectPortal = 'manager'; } else { redirectPortal = 'sales'; } } // Update Portal Badge switch (redirectPortal) { case 'super-admin': portalBadge.textContent = '🔐 SUPER ADMIN PORTAL'; portalBadge.style.background = '#dc3545'; portalBadge.style.color = 'white'; break; case 'manager': portalBadge.textContent = '📊 WAREHOUSE MANAGER PORTAL'; portalBadge.style.background = '#ffc107'; portalBadge.style.color = '#333'; break; default: portalBadge.textContent = '💰 SALES TEAM PORTAL'; portalBadge.style.background = '#16A34A'; portalBadge.style.color = 'white'; } // ============================================================ // 2. REMEMBER ME // ============================================================ if (localStorage.getItem('rememberUser')) { username.value = localStorage.getItem('rememberUser'); rememberMe.checked = true; } // ============================================================ // 3. PASSWORD TOGGLE // ============================================================ togglePassword.addEventListener('click', function() { const icon = this.querySelector('i'); if (password.type === 'password') { password.type = 'text'; icon.classList.remove('fa-eye'); icon.classList.add('fa-eye-slash'); } else { password.type = 'password'; icon.classList.remove('fa-eye-slash'); icon.classList.add('fa-eye'); } }); // ============================================================ // 4. CAPS LOCK DETECTION // ============================================================ password.addEventListener('keyup', function(e) { if (e.getModifierState('CapsLock')) { showInfo('⚠️ Caps Lock is ON'); } else { infoMsg.style.display = 'none'; } }); // ============================================================ // 5. DARK MODE // ============================================================ if (localStorage.getItem('theme') === 'dark') { document.body.classList.add('dark'); themeToggle.innerHTML = ''; } themeToggle.addEventListener('click', function() { document.body.classList.toggle('dark'); if (document.body.classList.contains('dark')) { localStorage.setItem('theme', 'dark'); this.innerHTML = ''; } else { localStorage.setItem('theme', 'light'); this.innerHTML = ''; } }); // ============================================================ // 6. MESSAGE FUNCTIONS // ============================================================ function showError(msg) { errorMsg.style.display = 'block'; infoMsg.style.display = 'none'; errorMsg.textContent = msg; } function showInfo(msg) { infoMsg.style.display = 'block'; errorMsg.style.display = 'none'; infoMsg.textContent = msg; } function clearMessages() { errorMsg.style.display = 'none'; infoMsg.style.display = 'none'; } // ============================================================ // 7. VALIDATION // ============================================================ function validate() { clearMessages(); if (username.value.trim() === '') { showError('Employee ID is required.'); username.focus(); return false; } if (password.value.trim() === '') { showError('Password is required.'); password.focus(); return false; } if (isTwoFAMode && twofaCode.value.trim() === '') { showError('2FA secret code is required.'); twofaCode.focus(); return false; } return true; } // ============================================================ // 8. LOGIN SUBMIT // ============================================================ loginForm.addEventListener('submit', async function(e) { e.preventDefault(); if (!validate()) return; loginBtn.classList.add('loading'); loginBtn.disabled = true; loginBtn.innerHTML = ' Authenticating...'; if (rememberMe.checked) { localStorage.setItem('rememberUser', username.value); } else { localStorage.removeItem('rememberUser'); } try { // 2FA Mode if (isTwoFAMode) { const response = await fetch(`${API_URL}/auth/verify-2fa`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ sessionId: twofaSessionId, secretCode: twofaCode.value.trim() }) }); const data = await response.json(); if (!data.success) { showError(data.message || '2FA verification failed'); loginBtn.classList.remove('loading'); loginBtn.disabled = false; loginBtn.innerHTML = ' Secure Sign In'; return; } if (data.token) { localStorage.setItem('token', data.token); localStorage.setItem('user', JSON.stringify(data.user)); showInfo('✅ Authentication successful! Redirecting...'); setTimeout(() => { window.location.href = data.redirectUrl; }, 1000); } return; } // Normal Login const response = await fetch(`${API_URL}/auth/login-portal`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: username.value.trim(), password: password.value.trim(), portal: redirectPortal }) }); const data = await response.json(); if (!data.success) { showError(data.message || 'Login failed. Please check your credentials.'); loginBtn.classList.remove('loading'); loginBtn.disabled = false; loginBtn.innerHTML = ' Secure Sign In'; return; } // 2FA Required if (data.requiresTwoFA && data.sessionId) { isTwoFAMode = true; twofaSessionId = data.sessionId; twofaGroup.classList.remove('hidden'); showInfo('🔐 Two-factor authentication required. Enter your secret code.'); loginBtn.innerHTML = ' Verify 2FA Code'; loginBtn.classList.remove('loading'); loginBtn.disabled = false; return; } // Login Success if (data.token) { localStorage.setItem('token', data.token); localStorage.setItem('user', JSON.stringify(data.user)); showInfo('✅ Authentication successful! Redirecting...'); setTimeout(() => { window.location.href = data.redirectUrl; }, 1000); } } catch (error) { console.error('Login error:', error); showError('Network error. Please check your connection.'); } finally { loginBtn.classList.remove('loading'); loginBtn.disabled = false; if (!isTwoFAMode) { loginBtn.innerHTML = ' Secure Sign In'; } } }); // ============================================================ // 9. ENTER KEY SUPPORT // ============================================================ document.addEventListener('keypress', function(e) { if (e.key === 'Enter') { loginForm.requestSubmit(); } }); // ============================================================ // 10. AUTO FOCUS // ============================================================ username.focus(); // ============================================================ // 11. PWA SERVICE WORKER // ============================================================ if ('serviceWorker' in navigator) { window.addEventListener('load', function() { navigator.serviceWorker.register('/sw.js') .then(function(reg) { console.log('Service Worker registered:', reg); }) .catch(function(err) { console.error('Service Worker registration failed:', err); }); }); } });
