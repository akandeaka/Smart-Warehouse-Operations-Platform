async function submitDaySales() { const today = new Date().toISOString().split('T')[0]; // First confirmation if (!confirm('📤 SUBMIT SALES FOR REVIEW\n\nAre you sure you want to submit today\'s sales for manager review?\n\nYou can still accept new allocations and continue selling after submission.\n\nClick OK to submit.')) { return; } try { // First API call - check for remaining goods const result = await fetchAPI('/sales/submit-day', { method: 'POST', body: JSON.stringify({ team_id: currentUser.team_id, sales_date: today }) }); if (result && result.success) { // No remaining goods - simple success showMessage('✅ Sales submitted to manager!', false, false, true); refreshData(); } else if (result && result.requires_action) { // There are remaining goods - show popup with TWO options const userChoice = confirm( `${result.message}\n\n` + `━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n` + `👉 Click OK to MOVE goods to warehouse\n` + `👉 Click CANCEL to KEEP goods for selling\n` + `━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━` ); // userChoice = true → OK button (Move goods) // userChoice = false → Cancel button (Keep goods) const finalResult = await fetchAPI('/sales/submit-day', { method: 'POST', body: JSON.stringify({ team_id: currentUser.team_id, sales_date: today, move_remaining_goods: userChoice // true = Move, false = Keep }) }); if (finalResult && finalResult.success) { if (userChoice === true) { // OPTION 1: Move goods to warehouse showMessage(`✅ Sales submitted! ${finalResult.message || 'Goods moved to warehouse.'}`, false, false, true); } else { // OPTION 2: Keep goods for continued selling showMessage(`✅ Sales submitted! ${finalResult.message || 'Goods kept for continued selling.'}`, false, false, true); } refreshData(); } else { showMessage('Failed to submit: ' + (finalResult?.message || 'Unknown error'), true); } } else { showMessage('Failed to submit sales: ' + (result?.message || 'Unknown error'), true); } } catch (error) { console.error('Submit day error:', error); showMessage('Error submitting sales: ' + error.message, true); } }
